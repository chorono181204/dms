            if (titleChanged || categoryChanged) {
                try {
                    const oldPath = existing.content;
                    if (fs.existsSync(oldPath)) {
                        // 1. Get correct root
                        const GDRIVE_ROOT = 'G:\\My Drive\\DMS';
                        const PREFERRED_ROOT = 'G:\\DMS_DATA';
                        const FALLBACK_ROOT = path.join(__dirname, '../../uploads');
                        
                        let root = FALLBACK_ROOT;
                        if (fs.existsSync('G:\\')) {
                            if (fs.existsSync(GDRIVE_ROOT)) root = GDRIVE_ROOT;
                            else if (fs.existsSync('G:\\My Drive')) root = GDRIVE_ROOT;
                            else if (fs.existsSync(PREFERRED_ROOT)) root = PREFERRED_ROOT;
                        }

                        // 2. Get Department Name
                        let departmentName = 'General';
                        const deptId = updateBody.departmentId ? parseInt(updateBody.departmentId) : existing.departmentId;
                        if (deptId) {
                            const dept = await prisma.department.findUnique({ where: { id: deptId } });
                            if (dept) departmentName = dept.name;
                        }

                        // 3. Get Category Path - Rebuild strictly from DB to ensure hierarchy
                        const categoryPathParts: string[] = [];
                        let currentCatId = updateBody.categoryId ? parseInt(updateBody.categoryId) : existing.categoryId;
                        
                        while (currentCatId) {
                            const cat = await prisma.category.findUnique({ where: { id: currentCatId } });
                            if (cat) {
                                categoryPathParts.unshift(cat.name);
                                currentCatId = cat.parentId || 0;
                            } else {
                                break;
                            }
                        }

                        // 4. Sanitize (Minimal)
                        const sanitize = (name: string) => name.replace(/[<>:"\\/\\|?*]/g, '_');
                        
                        const safeDept = sanitize(departmentName);
                        // If moving to root of Dept (no category)? logic handles empty array fine.
                        const safeCatPath = categoryPathParts.map(sanitize).join(path.sep);
                        
                        const targetDir = path.join(root, safeDept, safeCatPath);

                        if (!fs.existsSync(targetDir)) {
                            fs.mkdirSync(targetDir, { recursive: true });
                        }

                        // 5. Construct new filename
                        const ext = path.extname(oldPath);
                        const safeBaseName = sanitize(newTitle);
                        const newFilename = `${safeBaseName}_${Date.now()}${ext}`; // Consistent with saveDocumentFile

                        let newPath = path.join(targetDir, newFilename);
                        
                        // Rename / Move
                        if (oldPath !== newPath) {
                            try {
                                fs.renameSync(oldPath, newPath);
                                updateBody.content = newPath;
                            } catch (e: any) {
                                if (e.code === 'EXDEV') {
                                    fs.copyFileSync(oldPath, newPath);
                                    fs.unlinkSync(oldPath);
                                    updateBody.content = newPath;
                                } else {
                                    throw e;
                                }
                            }
                        }
                    }
                } catch (error) {
                    console.error('Error renaming/moving document file:', error);
                }
            }
